-- 3
-- Přirozený dotaz:
-- Vypiš výrobce a počet produktů, které dodávají, včetně průměrné ceny produktů.

SELECT
    v.nazev AS vyrobce,
    COUNT(*) AS pocet_produktu,
    ROUND(AVG(p.cena),2) AS prumerna_cena
FROM produkt p
JOIN vyrobce v
ON p.id_vyrobce = v.id_vyrobce
GROUP BY v.nazev;

-- 20
-- Přirozený dotaz:
-- Vypiš pouze ty výrobce, kteří nabízejí alespoň 2 produkty
-- a jejichž průměrná cena produktů je vyšší než 15000 Kč.
-- Výsledky seřaď od nejvyšší průměrné ceny.

SELECT
    v.nazev AS vyrobce,
    COUNT(p.id_produkt) AS pocet_produktu,
    ROUND(AVG(p.cena), 2) AS prumerna_cena
FROM vyrobce v
JOIN produkt p
ON v.id_vyrobce = p.id_vyrobce
GROUP BY v.nazev
HAVING COUNT(p.id_produkt) >= 2
   AND AVG(p.cena) > 15000
ORDER BY prumerna_cena DESC;

-- 24
-- Přirozený dotaz:
-- Vypiš všechny výrobce, počet jejich produktů, průměrnou cenu produktů
-- a cenu jejich nejdražšího produktu. Výsledky seřaď podle počtu produktů.

SELECT
    v.nazev AS vyrobce,
    COUNT(p.id_produkt) AS pocet_produktu,
    ROUND(AVG(p.cena), 2) AS prumerna_cena,
    MAX(p.cena) AS nejdrazsi_produkt
FROM vyrobce v
JOIN produkt p
ON v.id_vyrobce = p.id_vyrobce
GROUP BY v.nazev
ORDER BY pocet_produktu DESC;
